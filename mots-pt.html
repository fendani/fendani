function renderDetails(mot) { let col1HTML = ""; let col2HTML = ""; const lang = new URLSearchParams(window.location.search).get("lang") || "pt"; // Fonction pour récupérer le bon texte function t(value) { if (typeof value === "object" && value !== null) { return value[lang] || value.fr || ""; } return value || ""; } // DÉFINITION if (mot.definition) { col1HTML += `
Definição ${escapeHTML(t(mot.definition))}
`; } // EXEMPLE if (mot.example) { col1HTML += `
Exemplo ${escapeHTML(t(mot.example))}
`; } // NATURE + GENRE if (mot.type || mot.gender) { col2HTML += `
Natureza / Género ${escapeHTML(t(mot.type))} ${mot.gender ? " · " + escapeHTML(t(mot.gender)) : ""}
`; } // DOMAINE if (mot.domain) { col2HTML += `
Domínio ${escapeHTML(t(mot.domain))}
`; } // PRONONCIATION if (mot.phonetic) { col2HTML += `
Pronúncia ${escapeHTML(mot.phonetic)}
`; } // AUDIO if (mot.audio1 || mot.audio2) { col2HTML += `
Áudio`; if (mot.audio1) { col2HTML += ` `; } if (mot.audio2) { col2HTML += ` `; } col2HTML += `
`; } // ÉTYMOLOGIE if (mot.etymology) { const etymologies = Array.isArray(mot.etymology) ? mot.etymology : [mot.etymology]; etymologies.forEach(etymology => { const language = t(etymology.language); const meaning = t(etymology.meaning); if (etymology.source || meaning) { col2HTML += `
Etimologia · ${escapeHTML(language)} ${etymology.source || ""} ${meaning ? " — " + escapeHTML(meaning) : ""}
`; } }); } // RÉSONANCES if (mot.related && mot.related.length) { col2HTML += `
Palavras relacionadas ${mot.related.map(item => `${escapeHTML(item)}`).join("")}
`; } return `
${col1HTML}
${col2HTML}
`; }
